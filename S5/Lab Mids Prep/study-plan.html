🐍 Python OOP Template

class Animal:
    species = "Unknown"  # class var

    def __init__(self, name):
        self.name = name  # instance var

    def speak(self):  # instance method
        return f"I am {self.name}"

    @classmethod
    def info(cls):    # class method
        return cls.species

    @staticmethod
    def breathe():   # static method
        return "inhale"

class Dog(Animal):  # inheritance
    def speak(self):  # override = polymorphism
        return "Woof!"

🔍 BFS vs DFS in Python

from collections import deque

def bfs(graph, start):
    visited = set()
    queue = deque([start])
    while queue:
        node = queue.popleft()  # FIFO
        if node not in visited:
            visited.add(node)
            queue.extend(graph[node])
    return visited

def dfs(graph, start, visited=None):
    if visited is None: visited = set()
    visited.add(start)
    for nb in graph[start]:
        if nb not in visited:
            dfs(graph, nb, visited)
    return visited

⭐ A* Search Template

import heapq

def astar(graph, start, goal, h):
    pq = [(0, start)]  # (f, node)
    g = {start: 0}
    while pq:
        f, node = heapq.heappop(pq)
        if node == goal:
            return g[goal]
        for nb, cost in graph[node]:
            ng = g[node] + cost
            if nb not in g or ng < g[nb]:
                g[nb] = ng
                heapq.heappush(pq,
                    (ng + h[nb], nb))

# f(n) = g(n) + h(n)
# g = cost so far
# h = heuristic estimate to goal

🗄️ Star Schema SQL Pattern

-- Dimension with surrogate key
CREATE TABLE dim_product (
    product_sk   SERIAL PRIMARY KEY,
    product_nk   VARCHAR(50),
    product_name VARCHAR(100)
);
-- Fact with FK to dimension
CREATE TABLE fact_sales (
    sale_id    SERIAL PRIMARY KEY,
    product_sk INT REFERENCES dim_product,
    amount     NUMERIC(10,2),
    sale_date  DATE
);
-- ROLLUP / CUBE
SELECT region, SUM(amount)
FROM fact_sales f
JOIN dim_product p ON
  f.product_sk = p.product_sk
GROUP BY ROLLUP(region);

🔷 Databricks Medallion Pattern

# BRONZE — raw ingest
df = spark.read.json("/data/raw/*.json")
df.write.format("delta") \
    .mode("overwrite") \
    .save("/delta/bronze")

# SILVER — cleaned & typed
df_s = df.dropna() \
    .withColumn("ts",
        col("ts").cast("timestamp"))
df_s.write.format("delta") \
    .mode("append").save("/delta/silver")

# GOLD — business aggregation
df_g = df_s.groupBy("user_id") \
    .agg(count("*").alias("events"))
df_g.write.format("delta") \
    .mode("overwrite").save("/delta/gold")

🧠 Search Algorithm Comparison

Algorithm | Complete | Optimal | Space
----------|----------|---------|-------
BFS       |   Yes    |  Yes*   | O(b^d)
DFS       |   No†    |  No     | O(bm)
UCS       |   Yes    |  Yes    | O(b^d)
IDS       |   Yes    |  Yes*   | O(bd)
GBFS      |   No     |  No     | O(b^m)
A*        |   Yes    |  Yes**  | O(b^d)

* = unit step costs only
** = admissible heuristic
† = loops on cyclic graphs

Minimax: MAX picks max, MIN picks min
Alpha-Beta: prune when alpha >= beta
Hill Climbing: always go up — fails
  at local maxima, plateaus, ridges

📊 Key DW Concepts

OLTP vs DW:
  OLTP: current data, normalized,
        row-store, INSERT/UPDATE heavy
  DW:   historical, denormalized,
        column-store, SELECT/analytics

Grain: most atomic row in fact table
  e.g. "one row per sale line item"

Surrogate Key: system INT PK (SERIAL)
Natural Key: real-world ID (e.g. SKU)

ROLAP:  SQL on relational DB (flexible)
MOLAP:  pre-aggregated cubes (fastest)
HOLAP:  hybrid (detail ROLAP + agg MOLAP)
DOLAP:  download cube to desktop

Normalization: remove redundancy (3NF)
Denormalization: pre-join for speed
